public class MainLuz {
    public static void main(String[] args) {
        LuzInteligente[] luces = new LuzInteligente[2];

        luces[0] = new LuzInteligente("Zona_Sala");
        luces[1] = new LuzInteligente("Zona_Patio");

        luces[0].encender();
        luces[0].ajustarBrillo(75);
        luces[1].apagar();

        for (int i = 0; i < luces.length; i++) {
            luces[i].mostrarEstado();
        }
    }
}
