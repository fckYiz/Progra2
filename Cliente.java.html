public class Cliente {
    private String nombre;
    private String tipoCliente;
    private double totalCompra;

    public Cliente(String nombre, String tipoCliente, double totalCompra) {
        this.nombre = nombre;
        this.tipoCliente = tipoCliente;
        this.totalCompra = totalCompra;
    }

    public String getTipoCliente() { return tipoCliente; }
    public double getTotalCompra() { return totalCompra; }

    public double calcularDescuento() {
        if (tipoCliente.equalsIgnoreCase("premium")) return totalCompra * 0.20;
        else if (tipoCliente.equalsIgnoreCase("estándar")) return totalCompra * 0.10;
        return 0;
    }

    public void mostrarResumen() {
        double descuento = calcularDescuento();
        double finalPagar = totalCompra - descuento;
        System.out.println("Cliente: " + nombre + " (" + tipoCliente + ")");
        System.out.println("Subtotal: $" + totalCompra + " | Descuento: $" + descuento + " | Total final: $" + finalPagar);
        System.out.println("-------------------------------------------------");
    }
}
