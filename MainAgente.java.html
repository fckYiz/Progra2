import java.util.Scanner;

public class MainAgente {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        final int MAX = 5;
        Agente[] agentes = new Agente[MAX];
        int indice = 0;
        int opcion;

        do {
            System.out.println("\n1. Agregar Agente\n2. Modificar Energía por Índice\n3. Mostrar Agentes\n4. Salir");
            System.out.print("Opción: ");
            opcion = Integer.parseInt(sc.nextLine());

            if (opcion == 1) {
                if (indice < MAX) {
                    System.out.print("Nombre del agente: ");
                    String nombre = sc.nextLine();
                    System.out.print("Nivel de energía (0-100): ");
                    int energia = Integer.parseInt(sc.nextLine());
                    System.out.print("Estado (activo/inactivo): ");
                    String estado = sc.nextLine();

                    agentes[indice++] = new Agente(nombre, energia, estado);
                    System.out.println("Agente registrado.");
                } else {
                    System.out.println("No hay más memoria");
                }
            } else if (opcion == 2) {
                System.out.print("Ingrese el índice del agente a modificar (0 a " + (indice - 1) + "): ");
                int pos = Integer.parseInt(sc.nextLine());

                if (pos >= 0 && pos < indice) {
                    System.out.print("Nuevo nivel de energía: ");
                    int nuevaEnergia = Integer.parseInt(sc.nextLine());
                    agentes[pos].setNivelEnergia(nuevaEnergia);
                    System.out.println("Energía actualizada.");
                } else {
                    System.out.println("Índice no válido.");
                }
            } else if (opcion == 3) {
                for (int i = 0; i < indice; i++) {
                    System.out.print("Índice " + i + " - ");
                    agentes[i].mostrarEstado();
                }
            }
        } while (opcion != 4);
        sc.close();
    }
}
